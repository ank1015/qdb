# The benchmark command list.
#
# One command per line. Every command runs on the real GNU tools and on the
# engine, against the same data, and the results are compared.
#
# {NAME} is a placeholder. Its value differs per dataset and is kept in
# bench/terms/DATASET.txt: search terms chosen by the share of files that
# contain them, and a few paths.
#
#   {ABSENT}  a term in no file          {SOME}    a term in about 10% of files
#   {ONE}     a term in exactly one file {COMMON}  a term in over half the files
#   {RARE}    a term in about 0.1%       {RAREUP}  {RARE} in capitals
#   {FEW}     a term in about 1%         {PREFIX}  the start of many function names
#   {DIR}     a folder with about 15% of the files, {SMALLDIR} one with about 1%
#   {GLOB}    many files of one kind in one folder ({GLOBDIR}), {FILE} one of them
#   {EXT}, {EXT2}  the two most common file extensions
#
# Lines starting with "mut: " change the data. They run once, in order,
# and the list ends by removing everything it created. All other lines
# only read, and are timed over several runs.

# List
ls
ls -a
ls -R
ls -aR {DIR}
ls -p {DIR}
find .
find . -name '*.{EXT}'
find . -name '*.{EXT2}'
find . -type d
find . -type f
find . -maxdepth 2 -type f
find {DIR} -name '*.{EXT}' -type f
find . -name 'Makefile'

# Read
cat {FILE}
cat -n {FILE}
head -n 20 {FILE}
tail -n 20 {FILE}
tail -n +100 {FILE}
wc {FILE}
wc -l {GLOB}
cat {GLOB} | wc -l
cat {FILE} | head -n 50 | tail -n 10
grep -c {COMMON} {FILE}
grep -n {COMMON} {FILE}
sed 's/{COMMON}/XX/g' {FILE}
sed -E 's/([a-z]+)_([a-z]+)/\2_\1/g' {FILE}
cat {FILE} | sed 's/^/> /' | wc -c

# Search, one term from each frequency class: lines, then file names
grep -rn {ABSENT} .
grep -rl {ABSENT} .
grep -rn {ONE} .
grep -rl {ONE} .
grep -rn {RARE} .
grep -rl {RARE} .
grep -rn {FEW} .
grep -rl {FEW} .
grep -rn {SOME} .
grep -rl {SOME} .
grep -rn {COMMON} .
grep -rl {COMMON} .

# Search: other forms
grep -rc {FEW} .
grep -rin {RAREUP} .
grep -ri {FEW} .
grep -rn {FEW} {DIR}
grep -rl {SOME} {DIR}
grep -rF '{FEW}' .
grep -rE '{PREFIX}_[a-z]+\(' .
grep -rE '{RARE}|{ONE}' .
grep -rE '{SOME}|{FEW}' .
grep -rE '[0-9]{4,}' .
grep -rn '^#include' .
grep -rE '^(import|from) ' .
grep -rin 'return [a-z_]*;$' {DIR}
grep -rv e {SMALLDIR}
grep -r '\({FEW}\).*\1' .

# Search in files named on the command line
grep -n {RARE} {GLOB}
grep -l {FEW} {GLOB}
grep -c {ABSENT} {GLOB}

# Pipelines
grep -rl {SOME} . | wc -l
grep -rc {COMMON} . | grep -v ':0$' | wc -l
find . -name '*.{EXT}' | wc -l
find . -type f | grep -c {EXT}

# Pipelines that end in head
grep -rn {COMMON} . | head -n 5
grep -rn {SOME} . | head -n 1
find . -type f | head -n 3
cat {GLOB} | head -n 2

# Stage 4. Search with more flags
grep -rno {FEW} .
grep -rw {FEW} .
grep -rnw {RARE} .
grep -rn -C 2 {RARE} .
grep -rn -A 1 -B 1 {FEW} .
grep -rho '{PREFIX}_[a-z_]*' . | sort | uniq -c | sort -rn | head
grep -rhow '[a-z]*{RARE}[a-z]*' . | sort | uniq -c | sort -rn | head -n 5

# rg, against real rg on one thread. Both follow the ignore rules.
rg -j 1 {ABSENT}
rg -j 1 {ONE}
rg -j 1 {RARE}
rg -j 1 {FEW}
rg -j 1 {SOME}
rg -j 1 {COMMON}
rg -j 1 -l {FEW}
rg -j 1 -l {COMMON}
rg -j 1 -c {SOME}
rg -j 1 -i {RAREUP}
rg -j 1 -w {FEW}
rg -j 1 -F '{FEW}'
rg -j 1 -n -C 2 {RARE}
rg -j 1 '{PREFIX}_[a-z]+\('
rg -j 1 '{RARE}|{ONE}'
rg -j 1 '\b{RARE}\w*'
rg -j 1 -t c {FEW}
rg -j 1 -g '*.{EXT}' {FEW}
rg -j 1 {FEW} {DIR}
rg -j 1 -uu {FEW}
rg -j 1 --sort path {RARE}
rg -j 1 --files | wc -l
rg -j 1 -l {SOME} | wc -l

# Times and sizes
ls -ltR
ls -lt {GLOBDIR}
ls -ltr {DIR} | tail -n 3
find . -newer {FILE}
find . -newer {FILE} -name '*.{EXT}'
find . -type f -mtime +30
# Not compared: a real folder's own size depends on its history, and the
# engine counts every folder as 4096 bytes.
nocmp: du -s .
nocmp: du -sh {DIR}
du -sb .
du -sb {DIR}
du -b -d 1 {DIR}

# The new text commands
sort {FILE}
sort -r {FILE} | head -n 3
sort -u {FILE} | wc -l
sort -k 2 {FILE} | tail -n 3
cat {GLOB} | sort | uniq -c | sort -rn | head -n 5
uniq -c {FILE} | wc -l
cut -c 1-20 {FILE}
cut -d ' ' -f 1 {FILE} | sort | uniq | wc -l
tr a-z A-Z < {FILE}
tr -d '\n' < {FILE} | wc -c
tr -s ' ' < {FILE} | wc -c
nl {FILE}
nl -ba {FILE} | tail -n 3
seq 1 100000 | wc -l
seq -w 1 1000 | tail -n 2
basename {FILE}
basename -s .{EXT} {FILE}
dirname {FILE}
printf '%s\n' a b c
printf '%05d %-8s|%x\n' 42 name 255
test -f {FILE} && echo yes
[ -d {DIR} ] && echo folder
date -u -d '2024-02-29 12:00:00' '+%Y-%m-%d %H:%M:%S %s %a'
echo a b c | xargs -n 1 echo
find {SMALLDIR} -name '*.{EXT}' | xargs cat | wc -l
find . -name '*.{EXT}' | xargs grep -l {FEW} | wc -l

# The new shell syntax
echo {a,b,c}{1..3}
echo {1..20..3} {a..e}
echo $(wc -l < {FILE}) `basename {FILE}`
echo "first line: $(head -n 1 {FILE})"
wc -l < {FILE}
cat {FILE} > /dev/null 2>&1
cat qdb_no_such_file 2> /dev/null ; echo after
cat qdb_no_such_file 2>&1 | wc -l
grep -q {COMMON} {FILE} && echo found || echo none
false || echo fallback
true && grep -c {COMMON} {FILE}
cat <<EOF⏎one $(echo two)⏎three⏎EOF
wc -l <<'EOF'⏎a⏎b⏎EOF
echo one \⏎two
grep -c {COMMON} {FILE}⏎wc -l {FILE}

# Add, edit, delete

mut: echo 'another line' >> bench_out/hello.txt
mut: cp {FILE} bench_out/copy.txt
mut: cp -r {DIR} bench_out/dir_copy
mut: cp -r {GLOBDIR} bench_out/work
mut: sed -i 's/{COMMON}/QDBBENCHONE/g' bench_out/copy.txt
mut: sed -i -E 's/[0-9]+/N/g' bench_out/copy.txt
mut: sed -i 's/{COMMON}/QDBBENCHMANY/g' bench_out/work/*.{EXT}
mut: sed -i 's/{ABSENT}/QDBBENCHNONE/g' bench_out/work/*.{EXT}
mut: sed -i 's/{RARE}/QDBBENCHFEW/g' bench_out/work/*.{EXT}
mut: cp {FILE} bench_out/same.txt
mut: cp {FILE} bench_out/same.txt
mut: cat {FILE} > bench_out/same.txt
mut: grep -rc QDBBENCHMANY bench_out
mut: grep -rl QDBBENCHMANY .
mut: grep -rl QDBBENCHONE .
mut: cat {FILE} | tee bench_out/tee.txt | wc -l
mut: sort {FILE} > bench_out/sorted.txt
mut: sort -u {FILE} >> bench_out/sorted.txt
mut: cat qdb_no_such_file &> bench_out/both.txt
mut: cat qdb_no_such_file 2>> bench_out/both.txt

mut: cp bench_out/joined.txt bench_out/joined2.txt
mut: sed -i '1d' bench_out/joined2.txt
mut: wc -c bench_out/joined.txt bench_out/joined2.txt bench_out/tee.txt bench_out/sorted.txt
mut: cat bench_out/both.txt
mut: mv bench_out/dir_copy bench_out/dir_moved
mut: mv bench_out/hello.txt bench_out/a/b
mut: ls -R bench_out/a
mut: rm bench_out/a/b/hello.txt
mut: rmdir bench_out/a/b
mut: rm -r bench_out/dir_moved
mut: rm -rf bench_out
